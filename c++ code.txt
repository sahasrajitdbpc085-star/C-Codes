}    double getArea() const { return area; }    string getSeason() const { return season; }    double getYieldPerAcre() const { return yieldPerAcre; }    virtual double calculateTotalYield() const {        return area * yieldPerAcre;    }    virtual void displayDetails() const {        cout << left << setw(15) << name             << setw(10) << area             << setw(12) << season             << setw(15) << yieldPerAcre             << setw(15) << calculateTotalYield();    }    bool operator>(const Crop& other) const {        return this->calculateTotalYield() > other.calculateTotalYield();    }    static int getTotalCropCount() {        return totalCropCount;    }    virtual void saveToFile(ofstream& outFile) const {        outFile << name << "," << area << "," << season << "," << yieldPerAcre << ",";    }};int Crop::totalCropCount = 0;class CerealCrop : public Crop, public Advisory {private:    string fertilizerType;public:    CerealCrop() : Crop(), fertilizerType("Standard NPK") {}    CerealCrop(const string& n, double a, const string& s, double y, const string& fert)        : Crop(n, a, s, y), fertilizerType(fert) {}    string getFertilizerType() const { return fertilizerType; }    void displayDetails() const override {        Crop::displayDetails();        cout << setw(20) << fertilizerType << setw(20) << "N/A" << endl;    }    void giveAdvice() const override {        cout << "[Cereal Advisory - " << name << "]: Ensure balanced application of "             << fertilizerType << " during the tillering stage and maintain proper field drainage." << endl;    }    void saveToFile(ofstream& outFile) const override {        outFile << "CEREAL,";        Crop::saveToFile(outFile);        outFile << fertilizerType << "\n";    }};class VegetableCrop : public Crop, public Advisory {private:    int irrigationFrequency;public:    VegetableCrop() : Crop(), irrigationFrequency(2) {}    VegetableCrop(const string& n, double a, const string& s, double y, int freq)        : Crop(n, a, s, y) {        if (freq <= 0) throw invalid_argument("Irrigation frequency must be at least 1 day!");        irrigationFrequency = freq;    }    int getIrrigationFrequency() const { return irrigationFrequency; }    void displayDetails() const override {        Crop::displayDetails();        cout << setw(20) << "N/A"             << setw(20) << (to_string(irrigationFrequency) + " days") << endl;    }    void giveAdvice() const override {        cout << "[Vegetable Advisory - " << name << "]: Irrigate every "             << irrigationFrequency << " day(s) and monitor closely for fungal infections." << endl;    }    void saveToFile(ofstream& outFile) const override {        outFile << "VEGETABLE,";        Crop::saveToFile(outFile);        outFile << irrigationFrequency << "\n";    }};void saveAllCropsToFile(const vector<Crop*>& farm, const string& filename) {    ofstream outFile(filename);    if (!outFile) {        cerr << "Error opening file for writing!" << endl;        return;    }    for (const auto& crop : farm) {        crop->saveToFile(outFile);    }    outFile.close();    cout << "\n[Success] Crop data successfully saved to " << filename << endl;}void loadCropsFromFile(const string& filename) {    ifstream inFile(filename);    if (!inFile) {        cerr << "[Error] No saved file found (" << filename << ")." << endl;        return;    }    cout << "\n--- Retrieved Crop Data from File (" << filename << ") ---" << endl;    string line;    while (getline(inFile, line)) {        cout << line << endl;    }    inFile.close();}int main() {    vector<Crop*> farm;    int choice;    do {        cout << "\n=============================================\n";        cout << "      SMART FARM MANAGEMENT SYSTEM           \n";        cout << "=============================================\n";        cout << "1.
